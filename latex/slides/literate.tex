\section{The idea}

\begin{frame}{Write It for a Human}
  \begin{itemize}
  \item Knuth (1984): a program is a \txx{document} people read
  \item Explanation first, code as its consequence
  \item The starter code is cheap; understanding is the bottleneck
  \item So make the reasoning the main text
  \item Your report is already a literate layer over your scripts
  \end{itemize}
\end{frame}

\section{The format}

\begin{frame}[fragile]{A \texttt{.lit.md} Is Markdown Plus Two Rules}
  A \txx{named chunk} is defined once; prose explains it first:
\begin{verbatim}
## Load the CILI map

```python {#load-the-cili-map}
def load_cili() -> dict[str, str]:
    ...
```
\end{verbatim}
  A \txx{file chunk} is written to disk, and lists the pieces in order:
\begin{verbatim}
```python {file=wikidata_to_czech.py}
<<script-header>>
<<load-the-cili-map>>
<<query-wikidata>>
```
\end{verbatim}
\end{frame}

\begin{frame}[fragile]{Chunks and Diagrams}
  \begin{itemize}
  \item \verb|<<name>>| on its own line expands that chunk
  \item Indentation is preserved, so Python stays aligned
  \item A name used twice is joined, so prose can interleave a function
  \item A \verb|```mermaid| block is a diagram
  \end{itemize}
\begin{verbatim}
```mermaid
flowchart LR
  WD[Wikidata] -->|P5063| ILI --> CS[Czech synset]
```
\end{verbatim}
\end{frame}

\section{Tangle and weave}

\begin{frame}[fragile]{Tangle: Code Out}
  Pull the runnable script out of the document:
\begin{verbatim}
uv run ../litprog/tangle.py wikidata.lit.md
# -> wikidata_to_czech.py
uv run wikidata_to_czech.py 500
\end{verbatim}
  \begin{itemize}
  \item The \texttt{.py} is \txx{generated}: edit the \texttt{.lit.md}, not it
  \item Re-tangle after every edit, then run to check
  \item The tangler is $\sim$40 lines, read it too
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Weave: Document Out}
  Render the document a human reads:
\begin{verbatim}
uv run ../litprog/weave.py wikidata.lit.md
# -> wikidata.html  and  wikidata.pdf
\end{verbatim}
  \begin{itemize}
  \item Mermaid is rendered by \txx{Kroki} over HTTP: no node, no browser
  \item Pure Python $+$ \texttt{pandoc} $+$ \texttt{xelatex}
  \item GitHub renders the \texttt{.lit.md} with the diagram inline anyway
  \end{itemize}
\end{frame}

\section{The example}

\begin{frame}[fragile]{The Wikidata Example}
  \texttt{wikidata/wikidata.lit.md} reads as one story:
  \begin{itemize}
  \item The problem, then a diagram of the pipeline
  \item An assembly chunk listing the pieces
  \item One named chunk per piece, prose above each
  \item A real run at the end: \txx{show the output}
  \end{itemize}
  \bigskip
  It \txx{tangles} to \texttt{wikidata\_to\_czech.py} (passes \texttt{ruff}, runs
  live) and \txx{weaves} to \texttt{wikidata.html} / \texttt{.pdf}.
\end{frame}

\section{Your turn}

\begin{frame}{Try It}
  \begin{itemize}
  \item Open \texttt{wikidata/wikidata.lit.md} and read it top to bottom
  \item Change a chunk; re-tangle; run: see the \texttt{.py} update
  \item Weave it; open the PDF
  \item For your own project: one diagram, a few named chunks
  \item Keep the \texttt{.lit.md} the \txx{single source of truth}
  \end{itemize}
\end{frame}
