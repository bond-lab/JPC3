\section{Why be sure?}

\begin{frame}{Why be sure?}
  \begin{itemize}
  \item A wordnet is a \txx{trusted source}
  \item Many tools read it; few check it
  \item An error you add, everyone inherits
    \begin{itemize}
    \item e.g.\ a wrong hypernym copied into every OMW language
    \end{itemize}
  \item Once released, it is copied, and hard to recall
  \item The Czech wordnet feeds OMW, sense tagging, AAC, \ldots
  \end{itemize}
\end{frame}

\begin{frame}{The cost of one wrong link}
  \begin{itemize}
  \item Project 4 links pictographs for \txx{AAC} users
  \item A wrong synset tag \into\ the wrong picture
  \item A child who cannot speak is handed the wrong word
  \end{itemize}
\end{frame}

\section{The new bottleneck}

\begin{frame}{Understanding Is the Bottleneck}
  \begin{itemize}
  \item The starter code generates candidates in seconds
    \begin{itemize}
    \item feminine candidates for all 1{,}919 role-noun synsets at once
    \end{itemize}
  \item Machines already verify themselves well
  \item The slow part is \txx{understanding} what came out
  \item ``Cognitive debt'': skip it and you lose the plot
  \item Generation got cheap; understanding did not (Litt, 2026)
  \end{itemize}
  \vfill
  \scriptsize \url{https://www.geoffreylitt.com/2026/07/02/understanding-is-the-new-bottleneck.html}
\end{frame}

\begin{frame}{Staying in the Loop}
  Three ways to keep understanding ahead of the machine:
  \begin{itemize}
  \item Explain before you trust: write what the data \textit{says}
  \item Plot it: a chart shows the outliers a table hides
  \item \txx{Spot-check} a sample before trusting the whole
    \begin{itemize}
    \item judge 20 by hand before you generate 2{,}000
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Look at Your Data as You Go}
  \begin{itemize}
  \item Check the output at every step of the work
  \item Dump it in a \txx{viewable format}: TOML, CSV, a plot
  \item A wrong turn caught early is cheap to undo
  \item A literate program shows its output inline, by habit
  \end{itemize}
  \vfill
  \centering\small ``Actually look at your data!''
\end{frame}

\begin{frame}{Explainers as Literate Programming}
  \begin{itemize}
  \item Knuth (1984): write code for \txx{humans} to read
  \item Prose and code woven together, explanation first
  \item Litt's explainer is the same idea, authorship flipped
  \item The agent writes the code; you write the understanding
  \item Your report is the literate layer over the scripts
  \end{itemize}
  \vfill
  \scriptsize Revived for LLMs: ``Natural Language Outlines'' (2024),
  ``Verifiable Literate Programming'' for validating generated code (2026)
\end{frame}

\section{Building an explainer}

\begin{frame}{An Explainer, From the Code}
  \begin{itemize}
  \item Generate an HTML explainer \txx{from} the finished script
  \item Four parts: background, intuition, a code tour, a quiz
  \item One self-contained page, one link to share
  \item The script stays a normal file, with nothing to keep in sync
  \item The Wikidata script's explainer also grew a SPARQL primer and a glossary
  \end{itemize}
\end{frame}

\begin{frame}{Build It Interactively}
  \begin{itemize}
  \item React to a draft; each reply refines the same page
  \item The link stays stable, so you share it once
  \item Keep every version in the repo: \texttt{v0}, \texttt{v1}, \texttt{v2}
  \item The actual prompts behind this explainer:
    \begin{itemize}
    \item \texttt{v0}: ``build an explainer for this script''
    \item \texttt{v1}: ``where do the ILI mappings come from? add a SPARQL appendix''
    \item \texttt{v2}: ``add a glossary''
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Make It Repeatable}
  The same \txx{recipe} works for any project script:
  \begin{itemize}
  \item Point at a script: ask for background, intuition, code, quiz
  \item Read it, then ask for the next fix
  \item Keep each version; link the latest
  \end{itemize}
  \bigskip
  \centering\small Packaged as the \texttt{code-explainer} skill in the repo
\end{frame}

\section{Error analysis}

\begin{frame}{Error Analysis}
  \begin{itemize}
  \item Take a \txx{random} sample (aim for 200)
  \item Judge each by hand: right or wrong?
  \item Precision = correct / judged
  \item But the single number is the least of it
  \item Group the mistakes into \txx{types}: that is the payoff
  \item One real example per type, with a count
  \end{itemize}
\end{frame}

\begin{frame}{Running example: p\v{r}echylov\'an\'i}
  Candidate: add a feminine form to a role-noun synset.
  \begin{exe}
    \ex Czech [ces] \\
        \gll u\v{c}itel \quad\into\quad u\v{c}itel-ka \\
             teacher.\gm{m} \qquad teacher-\gm{f} \\
    \trans ``(male) teacher \into\ (female) teacher''
  \end{exe}
  \begin{itemize}
  \item 1{,}919 role-noun synsets to work from (\texttt{noun.person})
  \item Easy to generate; the question is which are \txx{right}
  \end{itemize}
\end{frame}

\begin{frame}{An Error Taxonomy}
  A sample of 200 generated feminine candidates:
  \begin{center}
  \begin{tabular}{lrl}
    \toprule
    Error type & Count & Example \\
    \midrule
    Correct                 & 171 & \textit{l\'eka\v{r} \into\ l\'eka\v{r}ka} \\
    Coincidental suffix     &  12 & \textit{past} (not a base $+$ \textit{-ka}) \\
    Wrong sense             &   9 & feminine of a non-personal reading \\
    Already present         &   8 & lemma was in the synset \\
    \bottomrule
  \end{tabular}
  \end{center}
  \bigskip
  \txx{Precision 0.86} --- but the four rows tell the next team where to look
\end{frame}

\section{Agreement}

\begin{frame}{Why Two Annotators?}
  \begin{itemize}
  \item One judge can be \txx{consistently} wrong
    \begin{itemize}
    \item e.g.\ always accepting a coincidental \textit{-ka} (\textit{past}\into\textit{pastka})
    \end{itemize}
  \item A second judge tests whether the scheme is learnable
  \item Agreement $\approx$ reliability of the guidelines
  \item You work in pairs, so measure it
  \item Report how you resolved the disagreements
  \end{itemize}
\end{frame}

\begin{frame}{Raw Agreement Overcounts}
  \begin{itemize}
  \item ``We agreed 85\% of the time'' sounds good
  \item But some agreement happens by \txx{chance}
  \item Two coin-flippers agree half the time
  \item If almost everything is ``correct'', agreeing is easy
  \item Correct for chance \into\ Cohen's $\kappa$
  \end{itemize}
\end{frame}

\section{Cohen's kappa}

\begin{frame}{Cohen's Kappa}
  Agreement above what chance would give:
  \begin{block}{}
    \centering\large
    $\kappa = \dfrac{p_o - p_e}{1 - p_e}$
  \end{block}
  \begin{itemize}
  \item $p_o$: observed agreement (the proportion you both called the same)
  \item $p_e$: expected agreement \txx{by chance}, from each annotator's rates
    \begin{itemize}
    \item if you both say ``good'' 90\% of the time, chance agreement is high
    \end{itemize}
  \item $\kappa = 1$ perfect; $\kappa = 0$ no better than chance
  \end{itemize}
\end{frame}

\begin{frame}{A Worked Example}
  200 candidates, two annotators, good vs.\ bad:
  \begin{center}
  \begin{tabular}{lrrr}
    \toprule
    & B: good & B: bad & total \\
    \midrule
    A: good & 130 & 20 & 150 \\
    A: bad  &  10 & 40 &  50 \\
    \midrule
    total   & 140 & 60 & 200 \\
    \bottomrule
  \end{tabular}
  \end{center}
  \begin{itemize}
  \item $p_o = (130+40)/200 = 0.85$
  \item $p_e = 0.75\!\times\!0.70 + 0.25\!\times\!0.30 = 0.60$
  \end{itemize}
  \begin{block}{}
    \centering $\kappa = \dfrac{0.85 - 0.60}{1 - 0.60} = \txx{0.63}$
  \end{block}
\end{frame}

\begin{frame}{Reading Kappa}
  \begin{itemize}
  \item Rough scale (Landis \& Koch, 1977): $>0.8$ very good, $0.6$--$0.8$ substantial
  \item Our 0.63 is \txx{substantial}: the guidelines mostly work
  \item Low $\kappa$? Fix the guidelines, then re-annotate
  \item Beware: few categories or a skewed split inflate it
  \item Report $\kappa$ together with the raw table
  \end{itemize}
\end{frame}

\section{Giving back}

\begin{frame}{Why Open? --- Not Just the Warm Inner Glow}
  \begin{itemize}
  \item Our goal: a better Czech wordnet, \emph{and} a \txx{contribution}
  \item Open data others can build on
  \item Your fixes reach every downstream user
  \item A result that outlives the course
  \end{itemize}
\end{frame}

\begin{frame}{Feedback Upstream}
  Every error you find is a \txx{gift} to the source.
  \begin{itemize}
  \item ARASAAC: report wrong synset tags; send Czech keywords back
  \item Wikidata: add or fix the ILI (P5063) and WordNet id (P8814)
  \item Global Symbols / ARAWN: contribute the matches you validated
  \item CILI: report synsets with no ILI; DeriNet: report gaps
  \item PDT: the unmapped residue is data for the maintainers
  \end{itemize}
\end{frame}

\begin{frame}{Give Back: ARASAAC}
  Project 4 --- pictographs for AAC.
  \begin{itemize}
  \item Found a \txx{wrong} synset tag? Report it to ARASAAC
  \item Missing Czech keyword? Submit the translation
  \item Note colour and gender variants you matched
    \begin{itemize}
    \item e.g.\ the generated xkcd swatch for \textit{chestnut}
    \end{itemize}
  \item Their pictograms carry synsets, so a fix helps every language
  \end{itemize}
\end{frame}

\begin{frame}{Give Back: Wikidata}
  Project 5 --- Wikidata \into\ ILI \into\ Czech.
  \begin{itemize}
  \item Only a WordNet id (P8814)? Add the \txx{ILI} (P5063)
  \item Linked to the wrong sense? Correct the statement
  \item Confirmed a Czech label? Add it, with a reference
    \begin{itemize}
    \item e.g.\ add \textit{Nizozemsko} to the \textit{Holandsko} synset
    \end{itemize}
  \item Cite the wordnet as the source: statements need one
  \end{itemize}
\end{frame}

\begin{frame}{Give Back: PDT and CILI}
  Project 3 --- the sense-tag residue.
  \begin{itemize}
  \item The \txx{unmapped} tags are a bug list for the maintainers
  \item Split or merged senses? Propose the current synset
  \item Share your 1.5 \into\ current mapping table
  \item New concept with no ILI? Ask CILI to mint one
  \end{itemize}
\end{frame}

\begin{frame}{Give Back: DeriNet}
  Project 2 --- derivational relations.
  \begin{itemize}
  \item Validated feminine-motion pairs? Offer them back
    \begin{itemize}
    \item e.g.\ \textit{u\v{c}itel}\into\textit{u\v{c}itelka}
    \end{itemize}
  \item A family the wordnet \txx{lacks}? Flag the gap
  \item A DeriNet error you caught? Report it
  \item Both resources improve from the same checking
  \end{itemize}
\end{frame}

\begin{frame}{The Virtuous Cycle}
  \begin{itemize}
  \item Open data gets better only when fixes \txx{flow back}
  \item A pull request or an issue beats a private fix
  \item Cite what you used; credit who made it
  \item Release under an open licence, loadable with \texttt{wn}
  \item Next year's class starts from your cleaner data
  \end{itemize}
  \bigskip
  \centering\small Every new wordnet makes the network richer
\end{frame}
